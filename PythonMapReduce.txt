import sys

counts = {}

for line in sys.stdin:
    line = line.strip()
    try:
        word, count = line.split('\t', 1)
        count = int(count)
        counts[word] = counts.get(word, 0) + count
    except ValueError:
        continue

for word, total in counts.items():
    print '%s\t%s' % (word, total)
